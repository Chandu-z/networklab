int isPalindrome(int n) {
    
    int temp = n,digit,rev=0;

    while (n > 0) {
        digit = n % 10;
        rev = rev * 10 + digit;
        n = n / 10;
    }

    if (rev == temp)
        printf("Palindrome\n");
    else
        printf("Not a Palindrome\n");

    return 1;
}
